enable
configure terminal

! Konfigurasi Port mengarah ke Core Switch
interface GigabitEthernet0/1
ip address 10.10.10.1 255.255.255.252
ip nat inside
no shutdown
exit

! Konfigurasi Port mengarah ke Internet/Cloud
interface GigabitEthernet0/0
ip address 8.8.8.1 255.255.255.0
ip nat outside
no shutdown
exit

! Static Routing agar Router tahu cara mengembalikan data ke jaringan lokal
ip route 192.168.0.0 255.255.0.0 10.10.10.2

! Konfigurasi NAT Overload (PAT)
access-list 1 permit 192.168.0.0 0.0.255.255
ip nat inside source list 1 interface GigabitEthernet0/0 overload
exit
